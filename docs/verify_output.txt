import ast, pathlib
files = sorted(pathlib.Path("commission").rglob("*.py"))
cnt = {"float_literal":0,"float_call":0,"Decimal_ref":0,"true_div":0,"round_call":0}
for f in files:
    t = ast.parse(f.read_text(encoding="utf-8"))
    for n in ast.walk(t):
        if isinstance(n, ast.Constant) and isinstance(n.value, float): cnt["float_literal"] += 1
        if isinstance(n, ast.Call) and isinstance(n.func, ast.Name):
            if n.func.id == "float": cnt["float_call"] += 1
            if n.func.id == "round": cnt["round_call"] += 1
        if isinstance(n, ast.Name) and n.id == "Decimal": cnt["Decimal_ref"] += 1
        if isinstance(n, ast.BinOp) and isinstance(n.op, ast.Div): cnt["true_div"] += 1
print("files:", ", ".join(str(f) for f in files))
for k, v in cnt.items(): print(f"{k:14s} = {v}")
